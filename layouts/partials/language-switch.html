{{- /* Language switcher. On a real page it must link to the TRANSLATION OF THE CURRENT PAGE, never to the home page. .AllTranslations includes the current page, so every entry resolves to the matching URL in the other language. Pages are ordered by the language weights set in config/_default/languages.*.toml (French first). The 404 page is the exception: Hugo builds it outside the content tree, so it has no translations and .AllTranslations is empty. Without a fallback the switcher simply vanished there — on the one page where a visitor who landed in the wrong language most needs it. In that case we point at each language's home, since the requested path exists in no language by definition. */ -}} {{- $current := .Page.Lang -}} {{- $items := slice -}} {{- with .AllTranslations -}} {{- if gt (len .) 1 -}} {{- range . -}} {{- $items = $items | append (dict "lang" .Language.Lang "code" .Language.LanguageCode "url" .Permalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $items -}} {{- range site.Sites -}} {{- $items = $items | append (dict "lang" .Language.Lang "code" .Language.LanguageCode "url" .Home.Permalink) -}} {{- end -}} {{- end -}} {{- if gt (len $items) 1 }}
{{ i18n "nav.language" }} {{- range $i, $t := $items }} {{- if $i }}/{{ end }} {{ $t.lang | upper }} {{- end }}
{{- end }}
